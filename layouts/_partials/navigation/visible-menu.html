{{- /* Returns the entries of a menu whose target page exists and is not an empty list page. Entries with a plain URL are always kept. */ -}} {{- $entries := slice -}} {{- range . -}} {{- $visible := true -}} {{- if .PageRef -}} {{- with site.GetPage .PageRef -}} {{- $visible = not (partial "helpers/is-empty-list.html" .) -}} {{- else -}} {{- $visible = false -}} {{- end -}} {{- end -}} {{- if $visible -}} {{- $entries = $entries | append . -}} {{- end -}} {{- end -}} {{- return $entries -}}
